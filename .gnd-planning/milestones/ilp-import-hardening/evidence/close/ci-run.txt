# IH-R3: the development workflow's steps, run locally on the PR head.
# Host: a second macOS host (Apple silicon), not this author's machine.
# Each Python version ran from its own clean git clone of d132003, at the same time.
# uv 0.10.10 (the build profile's pinned version), private to the run.
# Filtered for size, otherwise verbatim and sanitized: the --collect-only test-id
# listing and pytest progress-dot lines are dropped, with the count of dropped
# lines stated in place. Paths: <clone> is the clone, <tmp> the system temp dir,
# <home> the home directory, <user> the account name.
# The Linux leg of the workflow is not reproduced (IH-R3).

#### ci.sh
#!/bin/zsh
# usage: ci.sh PYTHON DIR -- mirrors .github/workflows/development.yml
PY=$1; D=$2; export UV_PYTHON=$PY
cd "$D" || exit 2
step() { echo "== $*"; "$@"; rc=$?; echo "== exit $rc"; [ $rc -eq 0 ] || FAIL=1; }
FAIL=0
echo "== candidate"
rm -rf candidate
step uv build --out-dir candidate/release
mkdir -p candidate/source && tar -xzf candidate/release/*.tar.gz -C candidate/source
step uv build --wheel candidate/source/* --out-dir "$PWD/candidate/comparison"
git rev-parse HEAD > candidate/SOURCE_COMMIT
(cd candidate && shasum -a 256 release/* comparison/* > SHA256SUMS && shasum -a 256 -c SHA256SUMS); echo "== exit $? (sha256 check)"
echo "== regression"
step uv sync --locked --group dev
step uv run gnd install --symlink --project --force
step uv run gnd doctor --project
step uv run pytest --collect-only -q
step uv run pytest -q -raP tests/test_agent_cli.py tests/test_guarded_corpus.py tests/test_public_souffle_successors.py tests/test_v70_phase3_successors.py --tb=short
step uv run pytest -q -ra --tb=short
echo "== installed"
step test "$(cat candidate/SOURCE_COMMIT)" = "$(git rev-parse HEAD)"
export GND_CANDIDATE_DIR="$PWD/candidate"; step uv run pytest -q -o addopts='' tests/test_release_acceptance.py -s --tb=short
[ $FAIL -eq 0 ] && echo "== CI RESULT python $PY: PASS" || echo "== CI RESULT python $PY: FAIL"
